{{- /* Fails the build when a content page's front matter is missing `title` or `description`, or either isn't text. Hugo itself only rejects YAML that doesn't parse, and blanks a `description` that isn't text before templates see it. Pages generated by a content adapter (.gotmpl) are skipped; the adapter sets their fields. */ -}} {{- with .File -}} {{- if eq .Ext "md" -}} {{- $path := .Path -}} {{- range slice "title" "description" -}} {{- $v := index $.Params . -}} {{- if or (not $v) (ne (printf "%T" $v) "string") -}} {{- errorf "content/%s: front matter: %q is required and must be text" $path . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
